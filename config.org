      '(("gnu" . "https://elpa.gnu.org/packages/")
        ("melpa" . "https://stable.melpa.org/packages/")))

*** Emacs                                                             :emacs:
#+begin_src elisp
  (use-package emacs
    :init
    (global-visual-line-mode 1)
    (global-auto-revert-mode 1)
    :bind (("M-SPC" . toggle-input-method))
    :config
    ;; Enable lexical binding
    (setq lexical-binding t)

    ;; Allow to use symlinks
    (setq find-file-visit-truename t)

    ;; Scroll step
    (setq scroll-step 1)

    ;; Line numbers display
    (setq line-number-display-limit large-file-warning-threshold
          line-number-display-limit-width 200)

    ;; Short confirm
    (setq use-short-answers t)

    ;; Default input method
    (setq default-input-method "russian-computer")

    ;; Tabs
    (setq tab-always-indent 'complete
          indent-tabs-mode nil)

    ;; Native compilation
    (setq native-comp-speed 2
          load-prefer-newer t)

    ;; Cursor
    (setq cursor-type 'bar)

    ;; Display buffer
    (setq display-buffer-alist
        '(("\\*Help"
             (display-buffer-same-window))
            (("\\*Warnings")
             (display-buffer-no-window))
            ("\\*Man"
             (display-buffer-same-window))
            ("\\*Async .*\\*"
             (display-buffer-no-window))))

    ;; Android-specific
    (when (and (= emacs-major-version 30) (android-p))
      (touch-screen-set-point-commands (mouse-set-point))
      (setq tool-bar-position 'bottom)
      (setq touch-screen-display-keyboard t))

    ;; Styling
    :preface
    (defvar font-height 90)
    (setq font-name "Monospace")
    :custom-face
    (default ((t (:family ,font-name :height ,font-height))))
    (fixed-pitch ((t (:family ,font-name :height ,font-height))))
    (fixed-pitch-serif ((t (:family ,font-name :height ,font-height))))
    (variable-pitch ((t (:family ,font-name :height ,font-height))))
    (variable-pitch-text ((t (:family ,font-name :height ,font-height))))
    )

*** Org-roam                                                       :org:roam:
#+begin_src elisp
(use-package org-roam
  :after org
  :hook (org-roam-post-node-insert . org-roam-db-sync)
  :bind (("C-c n i" . org-roam-node-insert)
         ("C-c n f" . org-roam-node-find)
         ("C-c r b" . org-roam-buffer-toggle)
         ("C-c r c" . org-roam-capture))
  :init
  (org-roam-db-autosync-mode 1)
  :config
  (let* ((dir "~/org-roam"))
    (unless (file-exists-p dir)
      (make-directory dir))

    ;; org-roam directory
    (setq org-roam-directory dir)

    ;; garbage collection
    (setq org-roam-db-gc-threshold most-positive-fixnum)

    ;; enable completion
    (setq org-roam-completion-everywhere t)

    ;; display buffer
    (add-to-list 'display-buffer-alist
                 '("\\*org-roam\\*"
                   (display-buffer-in-side-window)
                   (side . right)
                   (slot . 0)
                   (window-width . 0.33)
                   (window-parameters . ((no-other-window . t)))))

    ;; org-roam node display
    (setq org-roam-node-display-template
          (concat "${title:*} "
                  (propertize "${tags:20}" 'face 'org-tag)))

    ;; Disable automatic update of the database on save
    (setq org-roam-db-update-on-save nil)

    ;; org-roam capture templates
    (setq org-roam-capture-templates
          '(("n" "note" plain "%?"
             :target (file+head "%<%Y%m%d%H%M%S>-${slug}.org"
                                "#+title: ${title}\n")
             :unnarrowed t)))))
#+end_src
