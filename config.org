;; Initialize package.el
(require 'package)

;; Add package archives (e.g., MELPA)
(add-to-list 'package-archives '("melpa" . "https://melpa.org/packages/") t)
(add-to-list 'package-archives '("gnu" . "https://elpa.gnu.org/packages/") t)

(unless package-archive-contents
  (package-refresh-contents))

***** Emacs                                                           :emacs:
#+begin_src elisp
  (use-package emacs
    :ensure nil
    :init
    (global-visual-line-mode 1)
    (global-auto-revert-mode 1)
    :bind (("M-SPC" . toggle-input-method)
	   ("C-g" . keyboard-escape-quit)
	   ("C-. v" . view-mode)
	   )
    :config
    ;; Unbind keys
    (unbind-key "C-v")
    (unbind-key "<insert>")
    (unbind-key "<insertchar>")

    ;; Display buffer
    (setq display-buffer-alist
	  '(("\\*Help"
	     (display-buffer-same-window))
	    (("\\*Warnings")
	     (display-buffer-no-window))
	    ("\\*Man"
	     (display-buffer-same-window))
	    ("\\*Async .*\\*"
	     (display-buffer-no-window))))

    ;; Backup files
    (let* ((backup-dir (expand-file-name "backup" user-emacs-directory)))
      (setq kept-new-versions 2
	    kept-old-versions 2
	    delete-old-versions t
	    backup-directory-alist
	    `(("." . ,backup-dir))))

    ;; Android-specific
    (when (and (= emacs-major-version 30) (android-p))
      (touch-screen-set-point-commands (mouse-set-point))
      (setq tool-bar-position 'bottom)
      (setq touch-screen-display-keyboard t))

    )
#+end_src

***** Org-roam                                                     :org:roam:
#+begin_src elisp
(use-package org-roam
  :ensure t
  :hook ((org-roam-capture-new-node . org-roam-db-sync))
  :bind (("C-c n i" . org-roam-node-insert)
	 ("C-c n f" . org-roam-node-find)
	 ("C-c C-n" . org-roam-node-visit)
	 ("C-c r b" . org-roam-buffer-toggle)
	 ("C-c r c" . org-roam-capture)
	 )
  :init
  (let* ((dir (expand-file-name "roam" org-directory)))
    (unless (file-exists-p dir)
      (make-directory dir))
    (setq org-roam-directory dir))
  (org-roam-db-autosync-mode 1)
  :config
  ;; garbage collection
  (setq org-roam-db-gc-threshold most-positive-fixnum)

  ;; enable completion
  (setq org-roam-completion-everywhere t)

  ;; display buffer
  (add-to-list 'display-buffer-alist
	       '("\\*org-roam\\*"
		 (display-buffer-in-side-window)
		 (side . right)
		 (slot . 0)
		 (window-width . 0.33)
		 (window-parameters . ((no-other-window . t)))))

  ;; org-roam node display
  (setq org-roam-node-display-template
	(concat "${title:*} "
		(propertize "${tags:20}" 'face 'org-tag)))

  ;; Disable automatic update of the database on save
  (setq org-roam-db-update-on-save nil))
#+end_src
